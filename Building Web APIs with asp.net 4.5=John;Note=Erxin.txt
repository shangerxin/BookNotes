public class HeadersValueProvider: IValueProvider 
{
    private HttpRequestHeaders _headers;
    
    public HeadersValueProvider(HttpActionContext actionContext){
        if(actionContext == null){
            throw new ARgumentNullException("actionContent");
        }
        _headers = actionContext.Request.Headers;
    }
    
    public bool ContainsPrefix(string prefix){
        return _headers.Contains(prefix);
    }
    
    public ValueProviderResult GetValue(string key){
        IEnumerable<string> value;
        
        if(_headers.TryGetValues(key, out value))
        {
            string concatenatedValue = "";
            value.ToList().ForEach((s) -> concatenatedValue +=s);
            return New ValueProviderResult(value, concatenatedValue, CultureInfo.InvariantCulture);
        }
        return null;
    }
}

- OWIN pipline

middle ware is a class/function to manipulate the request, response. a self-contained component within the OWIN (Open Web Interface for .NET) pipeline that intercepts, inspects, and modifies incoming HTTP requests and outgoing HTTP responses.  web application from the underlying web server (such as IIS), allowing you to write cross-cutting concerns like logging, routing, and authentication in a modular, host-independent fashion

// Startup.cs 

OwinMiddleware.cs 

using Microsoft.Owin;
using System.Threading.Tasks;

public class RequestLoggingMiddleware : OwinMiddleware
{
    // The constructor receives a reference to the next middleware in line
    public RequestLoggingMiddleware(OwinMiddleware next) : base(next)
    {
    }

    // Invoke runs automatically on every HTTP request
    public override async Task Invoke(IOwinContext context)
    {
        // 1. Logic executed on the incoming request
        System.Diagnostics.Debug.WriteLine($"Inbound Request: {context.Request.Path}");

        // 2. Pass control to the next middleware component in the pipeline
        await Next.Invoke(context);

        // 3. Logic executed on the outgoing response (unwinding)
        System.Diagnostics.Debug.WriteLine($"Outbound Response Status: {context.Response.StatusCode}");
    }
}
